CREATE TABLE CUSTOMER (
    CUST_ID VARCHAR(10) PRIMARY KEY,
    CUST_NAME VARCHAR(100) NOT NULL,
    CUST_EMAIL VARCHAR(100) NOT NULL,
    CUST_PASS VARCHAR(100) NOT NULL,
    CUST_ADD VARCHAR(255),
    CUST_PHONE VARCHAR(20),
    USER_ROLE VARCHAR(20) DEFAULT 'Customer'
);

CREATE TABLE STAFF (
    STAFF_ID VARCHAR(10) PRIMARY KEY,
    STAFF_EMAIL VARCHAR(100) NOT NULL,
    STAFF_PASS VARCHAR(100) NOT NULL,
    STAFF_NAME VARCHAR(100) NOT NULL,
    STAFF_PHONE VARCHAR(20),
    USER_ROLE VARCHAR(20) DEFAULT 'Staff'
);

CREATE TABLE PRODUCT (
    PROD_ID VARCHAR(10) PRIMARY KEY,
    PROD_CODE VARCHAR(10) NOT NULL,
    PROD_NAME VARCHAR(100) NOT NULL,
    CATEGORY VARCHAR(50),
    TOTAL_QTY INT,
    PROD_STATUS VARCHAR(20),
    IMG_PATH VARCHAR(255),
    CREATE_DATE DATE  DEFAULT CURRENT_DATE,
    LAST_UPDATE DATE  DEFAULT CURRENT_DATE
);

CREATE TABLE VARIATION (
    VAR_ID VARCHAR(10) PRIMARY KEY,
    VAR_NAME VARCHAR(100),
    SKU VARCHAR(20),
    VAR_PRICE DECIMAL(10,2),
    QTY_BEFORE INT,    
    VAR_QTY INT,
    SOLD_QTY INT,
    VAR_STATUS VARCHAR(20),
    IMG_PATH VARCHAR(255),
    CREATE_DATE DATE DEFAULT CURRENT_DATE,
    LAST_UPDATE DATE DEFAULT CURRENT_DATE,
    PROD_ID VARCHAR(10) NOT NULL,

    CONSTRAINT fk_variation_product
        FOREIGN KEY (PROD_ID)
        REFERENCES PRODUCT(PROD_ID)
);

CREATE TABLE STOCK (
    STOCK_ID VARCHAR(10) PRIMARY KEY,
    STOCK_TYPE VARCHAR(10),
    STOCK_QTY INT,
    STOCK_DATE DATE,
    REMARK VARCHAR(255),
    VAR_ID VARCHAR(10),
    STAFF_ID VARCHAR(10),
    ORDER_ID VARCHAR(10),

    CONSTRAINT fk_stock_variation
        FOREIGN KEY (VAR_ID)
        REFERENCES VARIATION(VAR_ID),

    CONSTRAINT fk_stock_staff
        FOREIGN KEY (STAFF_ID)
        REFERENCES STAFF(STAFF_ID),

    CONSTRAINT fk_order_table
        FOREIGN KEY (ORDER_ID)
        REFERENCES ORDERS(ORDER_ID)
);

CREATE TABLE ORDERS (
    ORDER_ID VARCHAR(10) PRIMARY KEY,
    ORDER_NO VARCHAR(20),
    CUST_NO VARCHAR(20),
    SUB_TOTAL DECIMAL(10,2),
    STATUS VARCHAR(20) DEFAULT 'Pending',
    CUST_ID VARCHAR(10),

    CONSTRAINT fk_order_customer
        FOREIGN KEY (CUST_ID)
        REFERENCES CUSTOMER(CUST_ID)
);

CREATE TABLE ORDER_VAR (
    ORDERVAR_ID VARCHAR(10) PRIMARY KEY,
    ORDER_ID VARCHAR(10),
    VAR_ID VARCHAR(10),
    CUST_ID VARCHAR(10),
    PROD_QTY INT,
    TOTAL_PRICE DECIMAL(10,2),

    CONSTRAINT fk_ov_order
        FOREIGN KEY (ORDER_ID)
        REFERENCES ORDERS(ORDER_ID),

    CONSTRAINT fk_ov_variation
        FOREIGN KEY (VAR_ID)
        REFERENCES VARIATION(VAR_ID),

    CONSTRAINT fk_ov_customer
        FOREIGN KEY (CUST_ID)
        REFERENCES CUSTOMER(CUST_ID)

);

CREATE TABLE RECEIPT (
    RECEIPT_ID VARCHAR(10) PRIMARY KEY,
    PAY_TYPE VARCHAR(50),
    RECEIPT_DATE DATE,
    GST DECIMAL(10,2),
    GRAND_TOTAL DECIMAL(10,2),
    STAFF_ID VARCHAR(10),
    ORDER_ID VARCHAR(10),

    CONSTRAINT fk_receipt_staff
        FOREIGN KEY (STAFF_ID)
        REFERENCES STAFF(STAFF_ID),

    CONSTRAINT fk_receipt_order
        FOREIGN KEY (ORDER_ID)
        REFERENCES ORDERS(ORDER_ID)
);

SELECT v.VAR_NAME, v.SKU, SUM(CASE WHEN s.STOCK_TYPE = 'In' THEN s.STOCK_QTY ELSE 0 END) AS STOCK_IN,
SUM(CASE WHEN s.STOCK_TYPE = 'Out' THEN s.STOCK_QTY ELSE 0 END) AS STOCK_OUT, v.VAR_QTY, v.VAR_STATUS, v.VAR_ID 
FROM STOCK s 
JOIN VARIATION v ON s.VAR_ID = v.VAR_ID
JOIN PRODUCT p ON v.PROD_ID = p.PROD_ID 
GROUP BY v.VAR_ID, v.VAR_NAME, v.SKU, v.VAR_QTY, v.VAR_STATUS, v.VAR_ID
ORDER BY v.SKU;


SELECT r.RECEIPT_ID, r.RECEIPT_DATE, r.PAY_TYPE, r.GRAND_TOTAL, o.ORDER_NO, 
c.CUST_ID, o.CUST_NO, c.CUST_NAME, c.CUST_ADD, c.CUST_PHONE, c.CUST_EMAIL,
o.ORDER_ID
FROM RECEIPT r 
JOIN ORDERS o ON r.ORDER_ID = o.ORDER_ID
JOIN CUSTOMER c ON o.CUST_ID = c.CUST_ID
WHERE r.ORDER_ID = 'O001';

SELECT ov.ORDERVAR_ID, o.ORDER_ID, ov.VAR_ID, v.IMG_PATH, 
       p.PROD_NAME, v.VAR_NAME, ov.PROD_QTY, ov.TOTAL_PRICE, ov.CUST_ID
FROM ORDER_VAR ov
LEFT JOIN ORDERS o ON ov.ORDER_ID = o.ORDER_ID
JOIN VARIATION v ON ov.VAR_ID = v.VAR_ID
JOIN PRODUCT p ON v.PROD_ID = p.PROD_ID
WHERE ov.CUST_ID = 'C005' 
  AND ov.ORDER_ID IS NULL;
